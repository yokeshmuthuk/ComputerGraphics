#version 330

// Input from vertex shader
in vec3 FragPos;
in vec3 Normal;
in vec2 TexCoord;
in vec3 LightDir;
in vec3 ViewDir;

// Output color
out vec4 FragColor;

// Uniforms
uniform sampler2D earthTexture;
uniform vec3 lightColor;
uniform vec3 objectColor;      // Solid color for non-textured objects
uniform int useTexture;        // 1 = use texture, 0 = use objectColor

// Advanced Lighting Parameters (Phong Model)
uniform float ambientStrength;
uniform float diffuseStrength;
uniform float specularStrength;
uniform float shininess;

void main() {
    // Get base color (either from texture or solid color)
    vec3 baseColor;
    if (useTexture == 1) {
        baseColor = texture(earthTexture, TexCoord).rgb;
    } else {
        baseColor = objectColor;
    }

    // Normalize interpolated vectors
    vec3 norm = normalize(Normal);
    vec3 lightDir = normalize(LightDir);
    vec3 viewDir = normalize(ViewDir);

    // Ambient lighting
    vec3 ambient = ambientStrength * lightColor;

    // Diffuse lighting (Lambertian)
    float diff = max(dot(norm, lightDir), 0.0);
    vec3 diffuse = diffuseStrength * diff * lightColor;

    // Specular lighting (Phong)
    vec3 reflectDir = reflect(-lightDir, norm);
    float spec = pow(max(dot(viewDir, reflectDir), 0.0), shininess);
    vec3 specular = specularStrength * spec * lightColor;

    // Combine all lighting components with base color
    vec3 result = (ambient + diffuse + specular) * baseColor;

    FragColor = vec4(result, 1.0);
}
