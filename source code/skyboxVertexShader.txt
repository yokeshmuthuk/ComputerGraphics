#version 330

in vec3 vertex_position;

out vec3 TexCoords;

uniform mat4 view;
uniform mat4 proj;

void main() {
    TexCoords = vertex_position;

    // Remove translation from view matrix (keep only rotation)
    mat4 viewNoTranslation = mat4(mat3(view));

    vec4 pos = proj * viewNoTranslation * vec4(vertex_position, 1.0);

    // Set z to w so depth is always 1.0 (furthest possible)
    gl_Position = pos.xyww;
}
