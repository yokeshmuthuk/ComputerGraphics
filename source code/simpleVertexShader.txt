#version 330

// Vertex attributes
in vec3 vertex_position;
in vec3 vertex_normal;
in vec2 vertex_texcoord;

// Output to fragment shader
out vec3 FragPos;
out vec3 Normal;
out vec2 TexCoord;
out vec3 LightDir;
out vec3 ViewDir;

// Uniforms
uniform mat4 view;
uniform mat4 proj;
uniform mat4 model;
uniform vec3 lightPos;  // Light position in world coords

void main() {
    // Calculate fragment position in world space
    FragPos = vec3(model * vec4(vertex_position, 1.0));

    // Calculate normal in world space
    // For non-uniform scaling, should use transpose(inverse(mat3(model)))
    Normal = normalize(mat3(model) * vertex_normal);

    // Pass texture coordinates to fragment shader
    TexCoord = vertex_texcoord;

    // Calculate light direction (from fragment to light)
    LightDir = normalize(lightPos - FragPos);

    // Calculate view direction (from fragment to camera)
    vec3 viewPos = inverse(view)[3].xyz;  // Extract camera position from view matrix
    ViewDir = normalize(viewPos - FragPos);

    // Transform vertex position to clip space
    gl_Position = proj * view * vec4(FragPos, 1.0);
}
